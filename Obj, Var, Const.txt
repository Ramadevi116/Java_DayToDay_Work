--->Constructor : Is a method that initializes an object at the time of object creation.
->Same as class name.
->No return type.
->called automatically.(Default)

public class ExampleOne {
	int id;
	String name;

 public ExampleOne(){                       // Default Constructor

}
	
 public ExampleOne(int id, String name) {   // parameterized Constructor
	 this.id=id;
	 this.name=name;
	 
 }
 public void m1() {
	 System.out.println(id+" "+name);
	 
 }

}

public class MainClass {

	public static void main(String[] args) {
		ExampleOne exp=new ExampleOne(121,"hari"); 
		exp.m1();
	}

}
